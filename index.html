ربط Google Sheets وباقي التبويبات

قم بنشر التاب (File > Share > Publish to web) بصيغة CSV وادخل الرابط هنا للمزامنة الحية.

يسمح بالتحديث الفوري داخل الشيت الأصلي عندما ينقر الموظف على "حل التذكرة".

// كود Google Apps Script للتسميع المباشر في الشيت:

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.openById("171DXs48k4FOnpHGB155mLpvgRnXUxfsXoQT4AYGV8rA").getSheetByName(data.tabName || "W35");
  var rows = sheet.getDataRange().getValues();
  
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0] == data.matchId) { // Col A ID
      sheet.getRange(i + 1, 26).setValue("He's Right"); // Col Z Resolution Update
      sheet.getRange(i + 1, 29).setValue("Solved by " + data.user + ": " + data.resolutionNote); // Col AC
      return ContentService.createTextOutput(JSON.stringify({status: "SUCCESS"})).setMimeType(ContentService.MimeType.JSON);
    }
  }
}
