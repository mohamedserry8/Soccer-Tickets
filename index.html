AppsScript_Backend.gs 
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var matchId = data.matchId;
    var tabName = data.tabName || "W35";
    var actionNotes = data.actionNotes;
    var solvedBy = data.solvedBy;

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(tabName) || ss.getSheets()[0];
    var values = sheet.getDataRange().getValues();

    for (var i = 1; i < values.length; i++) {
      if (values[i][0].toString() === matchId.toString()) {
        sheet.getRange(i + 1, 28).setValue(actionNotes); // Column AB (Comment)
        sheet.getRange(i + 1, 26).setValue("He's Right"); // Column Z (Resolution)
        sheet.getRange(i + 1, 23).setValue(solvedBy);    // Column W (Responsible)
        return ContentService.createTextOutput(JSON.stringify({status: "success"}))
               .setMimeType(ContentService.MimeType.JSON);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status: "not_found"}));
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({error: err.toString()}));
  }
}
